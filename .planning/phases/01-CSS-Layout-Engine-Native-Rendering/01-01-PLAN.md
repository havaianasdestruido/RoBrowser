<task type="auto">
  <name>Task 2: Create StyleSheetLoader module</name>
  <files>src/shared/styles/StyleSheetLoader.luau</files>
  <action>
    Create `src/shared/styles/StyleSheetLoader.luau` module per D-10 (CSS engine outputs to StyleSheet instances).
    
    Module responsibilities:
    1. Load compiled StyleSheet modules (require pattern)
    2. Create StyleSheet instances via `StyleSheet.createStyleSheet()`
    3. Apply StyleSheets to GUI trees via StyleLink
    4. Tag GUI instances with CollectionService for class selector matching
    
    Implementation:
    ```luau
    --!strict
    --!optimize 2
    --!native
    
    -- StyleSheetLoader - Loads and applies rbx-css compiled Roblox StyleSheets
    -- Per D-01: Uses rbx-css to compile CSS to StyleSheet instances
    
    local StyleSheetLoader = {}
    
    local CollectionService = game:GetService("CollectionService")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    
    -- Load the compiled StyleSheet (generated by rbx-css)
    local function loadCompiledStyleSheet()
        local Shared = ReplicatedStorage:WaitForChild("Shared")
        local styles = Shared:WaitForChild("styles")
        local CompiledStyleSheet = require(styles:WaitForChild("CompiledStyleSheet"))
        return CompiledStyleSheet
    end
    
    -- Apply StyleSheet to a GUI tree via StyleLink
    function StyleSheetLoader.applyStyleSheet(guiTree: Instance, styleSheetModule)
        local sheet = styleSheetModule.createStyleSheet()
        local styleLink = Instance.new("StyleLink")
        styleLink.StyleSheet = sheet
        styleLink.Parent = guiTree
        return styleLink
    end
    
    -- Tag an instance for CSS class selector matching
    function StyleSheetLoader.tagInstance(instance: Instance, className: string)
        -- rbx-css uses "Class" attribute (per RESEARCH.md Example 2)
        instance:SetAttribute("Class", className)
        CollectionService:AddTag(instance, className)
    end
    
    -- Initialize: load compiled StyleSheet
    function StyleSheetLoader.init()
        local success, compiledSheet = pcall(loadCompiledStyleSheet)
        if success then
            StyleSheetLoader.compiledSheet = compiledSheet
            return compiledSheet
        end
        warn("StyleSheetLoader: Failed to load compiled StyleSheet")
        return nil
    end
    
    return StyleSheetLoader
    ```
    
    Follow project conventions:
    - Use `--!strict`, `--!optimize 2`, `--!native` headers
    - Use pcall for safe require (per CONVENTIONS.md)
    - Use camelCase for functions, PascalCase for module table
    - Run `stylua src/shared/styles/StyleSheetLoader.luau` to format
    - Run `selene src/shared/styles/StyleSheetLoader.luau` to lint
  </action>
  <verify>
    <automated>stylua --check src/shared/styles/StyleSheetLoader.luau && selene src/shared/styles/StyleSheetLoader.luau && test -f src/shared/styles/StyleSheetLoader.luau</automated>
  </verify>
  <done>StyleSheetLoader.luau created with init(), applyStyleSheet(), and tagInstance() methods. Module follows project conventions and passes stylua/selene checks.</done>
